# View Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Solid views of opencode-tamago idiomatic Solid (native reactive props, a theme context, one `Sprite`, one `DialogFrame`), testable headless under Bun with committed frame snapshots, and split the 435-line shell `index.tsx` into a `shell/` folder, with every user-facing string in `core/text/`.

**Architecture:** OpenCode compiles plugin JSX with `babel-preset-solid` (universal), so JSX props are getters and a component reads `props.x` reactively; the hand-rolled `() => T` accessor props are React reflexes and go away. `@opentui/solid` exports `testRender`, a headless renderer that captures the character frame; view and shell tests run under `bun test` (Bun compiles the Solid JSX; Node cannot), core and adapter tests stay under `node --test`. Order: tests first as the safety net, then props, then the theme context, then the shared components, then text, then the shell.

**Tech Stack:** TypeScript under Node 24 type stripping (core, adapter) and Bun 1.4.2 (view/shell tests, via mise), `solid-js` 1.9, `@opentui/solid` 0.4.5 (`testRender`, `useKeyboard`), `@opencode-ai/plugin/tui` types, `node:test` and `bun:test`.

**Spec:** the view-layer review of 2026-09-18 in the conversation: candidates 5, 1, 2+4, 3, 6+7; decisions: two runners (node for core/adapter, bun for view/shell); committed snapshots (`toMatchSnapshot`); user text in `core/text/` by surface. The controller reordered the theme context before `Sprite`/`DialogFrame` so those are born without a `theme` prop, and text before the shell so the shell imports it.

`view/__tests__/render.tsx`:

```tsx
/** @jsxImportSource @opentui/solid */
import { testRender, type JSX } from "@opentui/solid";

export type Size = { width: number; height: number };
/** Wide enough for the sidebar footer line, tall enough for a Bubble above the sprite. */
export const SIDEBAR: Size = { width: 44, height: 14 };
export const DIALOG: Size = { width: 60, height: 20 };

/** Renders once and returns the setup plus a `frame()` that renders again and captures the characters. */
export async function mount(node: () => JSX.Element, size: Size = DIALOG) {
  const setup = await testRender(node, { width: size.width, height: size.height });
  const frame = async (): Promise<string> => {
    await setup.renderOnce();
    return setup.captureCharFrame();
  };
  return { ...setup, frame };
}

/** One frame of `node`, trailing spaces stripped per line so snapshots stay readable. */
export async function frame(node: () => JSX.Element, size: Size = DIALOG): Promise<string> {
  const { frame: capture } = await mount(node, size);
  return trim(await capture());
}

export function trim(frame: string): string {
  return frame
    .split("\n")
    .map((line) => line.trimEnd())
    .join("\n")
    .replace(/\n+$/, "");
}
```

```tsx
export function Portrait(props: { lines: Frame; color: RGBA; bubble?: BubbleView; children: JSX.Element }): JSX.Element {
  /** Memoised so a tick that lands on the same cached frame re-renders nothing. */
  const lines = createMemo(() => props.lines);
  return (
    <box flexDirection="column">
      <Show when={props.bubble}>
        {(bubble) => (
          <box flexDirection="column">
            <text fg={bubble().border}>{bubble().top}</text>
            <text fg={bubble().border}>
              ( <span style={{ fg: bubble().ink }}>{bubble().text}</span> )
            </text>
            <text fg={bubble().border}>{bubble().bottom}</text>
          </box>
        )}
      </Show>
      <box flexDirection="row" gap={2}>
        <box flexDirection="column" flexShrink={0}>
          <Index each={lines()}>{(line) => <text fg={props.color}>{line()}</text>}</Index>
        </box>
        <box flexDirection="column" justifyContent="center">
          {props.children}
        </box>
      </box>
    </box>
  );
}
```

```ts
import { sameCareer, type Career } from "../core/career/career.ts";
import type { Session } from "../core/moment/session.ts";
import type { Voice } from "../core/speech/voice.ts";
import { tamago, type Tamago } from "../core/tamago.ts";
import type { Effect, Step, Window } from "../core/window.ts";

export type Mirror = {
  readonly current: () => Window;
  career: Accessor<Career>;
  sessions: Accessor<Record<string, Session>>;
  voices: Accessor<Record<string, Voice>>;
  muted: Accessor<boolean>;

  commit(next: Window): void;
  run(step: Step): void;

/**
 * The Window is the truth; core/window.ts moves it. Its parts are mirrored
 * into one signal each, so a view re-renders only for the part it reads.
 * `run` commits first, then performs the effects, so a toast reads the new Name.
 */
export function createMirror(initial: Window, defaultName: string, onEffect: (effect: Effect) => void): Mirror {
  let window = initial;
  const [career, setCareer] = createSignal<Career>(window.career, { equals: sameCareer });
  const [sessions, setSessions] = createSignal(window.sessions);
  const [voices, setVoices] = createSignal(window.voices);
  const [muted, setMuted] = createSignal(window.muted);
  const active = createMemo(() => tamago(career()));
  const name = () => career().name?.value ?? defaultName;
  const commit = (next: Window) => {
    const prev = window;

    if (next.voices !== prev.voices) setVoices(next.voices);

    current: () => window,
    career,
    sessions,
    voices,
    muted,
    active,
    name,
    commit,
    run({ window: next, effects }) {
      commit(next);
      for (const effect of effects) onEffect(effect);
    },
  };
}
```

```ts
export type Timers = { setTimeout: typeof setTimeout; clearTimeout: typeof clearTimeout };
const REAL: Timers = { setTimeout, clearTimeout };

/** The Tick loop: after each tick, waits `interval()` (fast while a session shows effort, slow otherwise). */
export function createTicker(deps: { interval: () => number; onTick: (now: number) => void; now?: () => number; timers?: Timers }): { start(): void; stop(): void } {
  const timers = deps.timers ?? REAL;
  const now = deps.now ?? Date.now;
  let handle: ReturnType<typeof setTimeout> | undefined;
  const schedule = () => {
    handle = timers.setTimeout(tick, deps.interval());
  };
  const tick = () => {
    deps.onTick(now());
    schedule();
  };
  return {
    start: schedule,
    stop() {
      if (handle !== undefined) timers.clearTimeout(handle);
    },
  };
}
```

```ts
import { backoff } from "../core/store/retry.ts";
import type { Timers } from "./tick.ts";

const REAL: Timers = { setTimeout, clearTimeout };

/** The Flush loop: `persist` every `base` ms; on a thrown disk error, back off up to `max` and count consecutive failures for `onError`. */
export function createFlushLoop(deps: { persist: () => boolean; onError: (err: unknown, failures: number) => void; base: number; max: number; timers?: Timers }): { start(): void; stop(): void } {
  const timers = deps.timers ?? REAL;
  let failures = 0;
  let handle: ReturnType<typeof setTimeout> | undefined;
  const schedule = () => {
    handle = timers.setTimeout(flush, backoff(failures, deps.base, deps.max));
  };
  const flush = () => {
    try {
      deps.persist();
      failures = 0;
    } catch (err) {
      failures += 1;
      deps.onError(err, failures);
    }
    schedule();
  };
  return {
    start: schedule,
    stop() {
      if (handle !== undefined) timers.clearTimeout(handle);
    },
  };
}
```

```ts
/** Fake timers: `fire()` runs the earliest pending callback. */
function fakeTimers() {
  const pending: { at: number; fn: () => void; id: number }[] = [];
  let id = 0;
  let clock = 0;
  const timers: Timers = {
    setTimeout: ((fn: () => void, ms: number) => {
      pending.push({ at: clock + ms, fn, id: ++id });
      return id as unknown as ReturnType<typeof setTimeout>;
    }) as typeof setTimeout,
    clearTimeout: ((handle: unknown) => {
      const at = pending.findIndex((entry) => entry.id === handle);
      if (at >= 0) pending.splice(at, 1);
    }) as typeof clearTimeout,
  };
  const fire = () => {
    pending.sort((a, b) => a.at - b.at);
    const next = pending.shift();
    if (next === undefined) throw new Error("nothing pending");
    clock = next.at;
    next.fn();
    return next.at;
  };
  return { timers, fire, pending };
}
```

```tsx
import type { TuiPlugin, TuiPluginModule } from "@opencode-ai/plugin/tui";
import type { Event } from "@opencode-ai/sdk/v2";
import { homedir } from "node:os";
import { join } from "node:path";

import { createStore, type Loaded } from "./adapter/store.ts";

import { createDialogs } from "./shell/dialogs.tsx";
import { createFlushLoop } from "./shell/flush.ts";
import { createGuard } from "./shell/guard.ts";

import { createPalette } from "./shell/palette.ts";
import { registerSlots } from "./shell/slots.tsx";
import { createTicker } from "./shell/tick.ts";

const id = "opencode-tamago";
const DATA_DIR = join(homedir(), ".local", "share", "opencode-tamago");
const FLUSH_MS = 2_000;
const FLUSH_MAX_MS = 60_000;

const tui: TuiPlugin = async (api, options) => {
  const defaultName = typeof options?.name === "string" && options.name.trim() ? options.name.trim() : "Tamago";
  const store = createStore(DATA_DIR);
  const logError = createErrorLog(DATA_DIR);
  const guard = createGuard(logError);
  const translate = createTranslator({ isChild: (id) => typeof api.state.session.get(id)?.parentID === "string" });

  let loaded: Loaded;
  try {
    loaded = store.load();
  } catch (err) {
    logError(err);
    loaded = { career: freshCareer(Date.now()), corrupt: false, present: false };
  }

  try {
    const started = Date.now();
    const [clock, setClock] = createSignal(0);
    let palette: ReturnType<typeof createPalette> | undefined;

    const mirror = createMirror(freshWindow(loaded.career, api.kv.get<boolean>("tamago.muted", false) === true), defaultName, (effect) => {
      const name = mirror.name();
      if (effect.type === "renamed") palette?.register();
      else if (effect.type === "switched") {
        palette?.register();
        api.ui.toast({ variant: "info", title: name, message: stepsIn(mirror.career(), defaultName) });
      } else if (effect.stage === "hatchling") api.ui.toast({ variant: "success", title: name, message: reveal(name, mirror.active()) });
      else api.ui.toast({ variant: "success", title: name, message: evolved(name, effect.stage) });
    });

    let warnedCorrupt = false;
    const warnCorrupt = () => {
      if (warnedCorrupt) return;
      warnedCorrupt = true;
      api.ui.toast({ variant: "warning", title: mirror.name(), message: CORRUPT });
    };
    if (loaded.corrupt) warnCorrupt();

    const actions = createActions({ api, store, mirror, warnCorrupt, guard });
    const dialogs = createDialogs({ api, store, mirror, actions, clock, started, defaultName, warnCorrupt, guard });
    palette = createPalette({ api, mirror, actions, dialogs, guard });
    palette.register();

    const onEvent = guard((event: Event) => {
      for (const addressed of translate(event)) mirror.run(receive(mirror.current(), addressed, Date.now()));
    });
    for (const type of SUBSCRIBED) api.lifecycle.onDispose(api.event.on(type, onEvent));

    const ticker = createTicker({
      interval: () => tickInterval(Object.values(mirror.current().sessions).map((session) => session.activity), behavior(mirror.current().career)),
      onTick: guard((now: number) => {
        mirror.commit(tickWindow(mirror.current(), now));
        setClock(now - started);
      }),

    ticker.start();

    const flusher = createFlushLoop({
      persist: actions.persist,
      onError: (err, failures) => {
        logError(err);
        if (failures === WARN_AFTER) api.ui.toast({ variant: "error", title: mirror.name(), message: cannotSave(mirror.name(), DATA_DIR) });
      },
      base: FLUSH_MS,
      max: FLUSH_MAX_MS,
    });
    flusher.start();

    const footer = (sessionID: string) => {
      const info = api.state.session.get(sessionID);
      const dir = info?.directory || api.state.path.directory;
      const branch = info?.directory === api.state.path.directory ? api.state.vcs?.branch : undefined;
      return { ...footerPath(dir, homedir(), branch), version: api.app.version };
    };
    registerSlots({ api, mirror, actions, clock, footer });

    api.lifecycle.onDispose(
      guard(() => {
        ticker.stop();
        flusher.stop();
        actions.dispose();
        palette?.dispose();
        const window = mirror.current();
        if (!isEmpty(window.pending)) store.flush(window.pending, window.career.hatchedAt);
      }),
    );
  } catch (err) {
    logError(err);
    return;
  }
};

const plugin: TuiPluginModule & { id: string } = { id, tui };
export default plugin;
```
